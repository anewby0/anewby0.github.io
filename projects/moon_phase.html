Moon Phase OLED

Fetch/Display Moon Data via ESP32+SSD1306

Moon phases are pretty neat. I often find myself wondering what the phase is while lacking lunar LOS (and googling it feels lame to be honest). With the flick of a switch, this device boots up and fetches that info for you; it also displays some bonus information and even some custom bitmap graphics.

Process

Here are my original designs for the project aswell the steps I followed.

"I want to design a system that fetches and displays accurate moon phase data using an ESP32 and SSD1306 OLED. It should be battery powered and have a power toggle."

  • Aquire ESP32 Devkit
  • Investigate fetching using http.GET()
  • Find and parse json
  • Set up SSD1306 to display data
  • Find json based on current time
  • Create and use custom bitmaps
  • Connect to batteries, reduce power usage

void fetchData() {
  //Check WiFi connection status
  if(WiFi.status()== WL_CONNECTED){
    HTTPClient http;

    String serverPath = "https://aa.usno.navy.mil/api/rstt/oneday?date=" + getDateString() + "&coords=35.65,-97.48&tz=-6";
    
    // Your Domain name with URL path or IP address with path
    http.begin(serverPath.c_str());
    
    // Send HTTP GET request
    int httpResponseCode = http.GET();
    
    if (httpResponseCode>0) {
      Serial.print("HTTP Response code: "); //200 means OK
      Serial.println(httpResponseCode);
      JsonDocument doc;
      deserializeJson(doc, http.getString());
      phase = String(doc["properties"]["data"]["curphase"]); //get current phase
      fracillum = String(doc["properties"]["data"]["fracillum"]); //fraction illuminated
      moon_event_0_name = String(doc["properties"]["data"]["moondata"][0]["phen"]).charAt(0);
      moon_event_0_time = String(doc["properties"]["data"]["moondata"][0]["time"]);
      moon_event_1_name = String(doc["properties"]["data"]["moondata"][1]["phen"]).charAt(0);
      moon_event_1_time = String(doc["properties"]["data"]["moondata"][1]["time"]);
      moon_event_2_name = String(doc["properties"]["data"]["moondata"][2]["phen"]).charAt(0);
      moon_event_2_time = String(doc["properties"]["data"]["moondata"][2]["time"]);
    }
    else {
      Serial.print("Error code: ");
      Serial.println(httpResponseCode);
    }
    // Free resources
    http.end();
  }
  else {
    Serial.println("WiFi Disconnected");
  }
}

void loop() {
  if ((millis() - lastTime) > timerDelay) {
    Serial.println("------");
    Serial.print("ENTERING LOOP AT ");
    Serial.println(millis()/1000);
    connectToWifi();
    fetchData();
    runDisplay();
    WiFi.disconnect();
    Serial.println("Disconnecting from Wifi network.");
    Serial.println("------");
    Serial.println("Sleeping...");
    lastTime = millis();
  }
}

void runDisplay() {
  Serial.println("Current Phase: " + phase);
  display.setCursor(20, 0);
  display.clearDisplay();
  display.setTextSize(1);
  display.println(phase);
  display.setTextSize(1);
  display.println(getDateString());
  display.println(fracillum);
  display.println(moon_event_0_name + ": " + moon_event_0_time);
  display.println(moon_event_1_name + ": " + moon_event_1_time);
  display.println(moon_event_2_name + ": " + moon_event_2_time);
  displayIcon();
  display.display();
  Serial.println("Display Set");
}

void displayIcon()
{
  if (phase == "Full Moon"){
    display.drawBitmap(75, 25, full_moon_icon, 32, 32, SSD1306_WHITE);
  }
  else if (phase == "New Moon"){
    display.drawBitmap(75, 25, new_moon_icon, 32, 32, SSD1306_WHITE);
  }
  else if (phase == "Waning Crescent"){
    display.drawBitmap(75, 25, waning_cres_icon, 32, 32, SSD1306_WHITE);   
  }
  else if (phase == "Waxing Crescent"){
    display.drawBitmap(75, 25, wax_cres_icon, 32, 32, SSD1306_WHITE);  
  }
  else if (phase == "Waning Gibbous"){
    display.drawBitmap(75, 25, waning_gib_icon, 32, 32, SSD1306_WHITE);  
  }
  else if (phase == "Waxing Gibbous"){
    display.drawBitmap(75, 25, wax_gib_icon, 32, 32, SSD1306_WHITE);  
  }
  else if (phase == "First Quarter"){
    display.drawBitmap(75, 25, first_quarter_icon, 32, 32, SSD1306_WHITE); 
  }
  else if (phase == "Last Quarter"){
    display.drawBitmap(75, 25, third_quarter_icon, 32, 32, SSD1306_WHITE);
  }
}

String getDateString()
{
  struct tm timeinfo;
  if(!getLocalTime(&timeinfo)){
    Serial.println("Failed to obtain time");
    return "2000-01-01";
  }
  //Serial.println(&timeinfo, "%A, %B %d %Y %m %H:%M:%S");
  char buffer[11];
  strftime(buffer, sizeof(buffer), "%F", &timeinfo);
  return buffer;
}

String getMonthDayString()
{
  struct tm timeinfo;
  if(!getLocalTime(&timeinfo)){
    Serial.println("Failed to obtain time");
    return "time-fail!";
  }
  //Serial.println(&timeinfo, "%A, %B %d %Y %m %H:%M:%S");
  char buffer[11];
  strftime(buffer, sizeof(buffer), "%F", &timeinfo);
  return buffer;
}

void connectToWifi()
{
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD); //gets it from secrets.h
  Serial.println("Connecting");
  while(WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("");
  Serial.print("Connected to WiFi network with IP Address: ");
  Serial.println(WiFi.localIP());
}
